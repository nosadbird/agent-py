public class MaxNumberUnderN {
    public static String findMaxNumber(int n, int[] digits) {
        String nStr = String.valueOf(n);
        int len = nStr.length();
        // 找出digits中的最大数字
        int maxDigit = Arrays.stream(digits).max().orElse(0);
        char[] result = new char[len];
        boolean tight = true;
        for (int i = 0; i < len; i++) {
            if (!tight) {
                result[i] = (char)('0' + maxDigit);
                continue;
            }
            char currentChar = nStr.charAt(i);
            int currentDigit = currentChar - '0';
            // 找出digits中≤currentDigit的最大数字
            int candidate = -1;
            for (int d : digits) {
                if (d <= currentDigit && d > candidate) {
                    candidate = d;
                }
            }
            if (candidate == -1) {
                // 需要回溯
                // 从i-1开始，尝试减小前一位
                int j = i - 1;
                while (j >= 0) {
                    // 尝试将第j位的数字减小
                    int prevDigit = result[j] - '0';
                    // 找出比prevDigit小的最大数字
                    int newDigit = -1;
                    for (int d : digits) {
                        if (d < prevDigit && d > newDigit) {
                            newDigit = d;
                        }
                    }
                    if (newDigit != -1) {
                        result[j] = (char)('0' + newDigit);
                        // 剩下的位填maxDigit
                        for (int k = j + 1; k < len; k++) {
                            result[k] = (char)('0' + maxDigit);
                        }
                        return new String(result);
                    }
                    j--;
                }
                // 无法构造len位数，返回len-1位
                char[] shorter = new char[len - 1];
                Arrays.fill(shorter, (char)('0' + maxDigit));
                return new String(shorter);
            }
            result[i] = (char)('0' + candidate);
            if (candidate < currentDigit) {
                tight = false;
            }
        }
        return new String(result);
    }
}

public class MaxNumber {
    public static int findMaxNumber(int n, int[] a) {
        // 去重并排序
        Set<Integer> set = new HashSet<>();
        for (int num : a) {
            set.add(num);
        }
        List<Integer> digits = new ArrayList<>(set);
        Collections.sort(digits, Collections.reverseOrder());

        String s = String.valueOf(n);
        int len = s.length();

        // 尝试构造长度为len的数字
        StringBuilder sb = new StringBuilder();
        boolean[] found = new boolean[1];
        if (construct(sb, s, digits, 0, found)) {
            return Integer.parseInt(sb.toString());
        }

        // 构造长度为len-1的最大数
        if (len > 1) {
            char maxDigit = (char)('0' + digits.get(0));
            return Integer.parseInt(String.valueOf(maxDigit).repeat(len - 1));
        } else {
            // len == 1，无法构造更短的数，返回-1或0？
            return -1;
        }
    }

    private static boolean construct(StringBuilder sb, String s, List<Integer> digits, int index, boolean[] found) {
        if (index == s.length()) {
            found[0] = true;
            return true;
        }

        char currentChar = s.charAt(index);
        int currentDigit = currentChar - '0';

        for (int d : digits) {
            if (d > currentDigit) continue;
            if (d < currentDigit) {
                sb.append(d);
                // 剩下的位填最大值
                for (int i = index + 1; i < s.length(); i++) {
                    sb.append(digits.get(0));
                }
                found[0] = true;
                return true;
            }
            // d == currentDigit
            sb.append(d);
            if (construct(sb, s, digits, index + 1, found)) {
                return true;
            }
            sb.setLength(sb.length() - 1); // 回溯
        }

        return false;
    }
}
